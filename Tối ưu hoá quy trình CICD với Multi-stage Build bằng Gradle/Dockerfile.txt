# ==========================================
# STAGE 1: Môi trường biên dịch (Builder)
# ==========================================
FROM gradle:7.6-jdk17 AS builder

# Thiết lập thư mục làm việc theo đúng đường dẫn yêu cầu
WORKDIR /home/gradle/project

# Copy toàn bộ mã nguồn vào vùng làm việc của container
COPY . .

# Cấp quyền thực thi cho Gradle wrapper và chạy lệnh đóng gói (bỏ qua test)
RUN chmod +x gradlew && ./gradlew clean build -x test

# ==========================================
# STAGE 2: Môi trường thực thi (Runtime)
# ==========================================
FROM eclipse-temurin:17-jre-alpine

# Thiết lập thư mục làm việc cho runtime
WORKDIR /app

# CHỈ copy duy nhất file .jar đã được biên dịch từ Stage 1 sang Stage 2
COPY --from=builder /home/gradle/project/build/libs/*.jar app.jar

# Mở cổng giao tiếp (Mặc định thường là 8080 cho Spring Boot)
EXPOSE 8080

# Lệnh khởi chạy ứng dụng Spring Boot
ENTRYPOINT ["java", "-jar", "app.jar"]
